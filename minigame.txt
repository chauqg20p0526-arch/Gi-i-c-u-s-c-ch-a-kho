            isTimerRunning = true;

        function updateTimer() {
            if (!isTimerRunning) return;

            if (timerSeconds <= 0) {
                clearInterval(timerInterval);
                document.getElementById('game-timer').textContent = "00:00";
                document.getElementById('timer-status-text').textContent = "ĐÃ HẾT GIỜ THI ĐẤU!";
                document.getElementById('game-timer').classList.add('text-red-500', 'animate-bounce');
                return;
            }

            timerSeconds--;
            const mins = Math.floor(timerSeconds / 60);
            const secs = timerSeconds % 60;
            const timeFormatted = `${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
            
            document.getElementById('game-timer').textContent = timeFormatted;

            // Update timer progress bar
            const pct = (timerSeconds / (15 * 60)) * 100;
            document.getElementById('timer-progress').style.width = pct + '%';

            // Warning colors when low on time
            if (timerSeconds < 180) { // < 3 mins
                document.getElementById('game-timer').className = "text-4xl sm:text-5xl font-black tracking-tight font-mono text-red-400 animate-pulse";
                document.getElementById('timer-progress').className = "bg-red-500 h-full w-full transition-all duration-1000";
            }
        }

        function toggleTimer() {
            isTimerRunning = !isTimerRunning;
            const icon = document.getElementById('timer-icon');
            if (isTimerRunning) {
                icon.className = 'fa-solid fa-pause';
                document.getElementById('timer-status-text').textContent = "Đồng hồ đang chạy...";
            } else {
                icon.className = 'fa-solid fa-play';
                document.getElementById('timer-status-text').textContent = "Đã tạm dừng đồng hồ";
            }
        }

        function resetTimer() {
            timerSeconds = 15 * 60;
            isTimerRunning = true;
            document.getElementById('game-timer').className = "text-4xl sm:text-5xl font-black tracking-tight font-mono text-amber-400 drop-shadow";
